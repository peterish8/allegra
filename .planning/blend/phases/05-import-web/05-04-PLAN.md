```ts
export interface ImportPlanInput {
  readonly source: 'spotify-export' | 'csv';
  readonly liked: readonly SongSnapshot[];                       // accepted matches only
  readonly playlists: readonly { name: string; songs: readonly SongSnapshot[] }[];
  readonly alreadyLiked: ReadonlySet<string>;                    // refs
  readonly now: number;
}
export async function playlistIdFor(source: string, name: string): Promise<string>; // 'import-' + 12 hex
export async function planImportOps(input: ImportPlanInput): Promise<{ chunks: LibraryOp[][]; already: number }>;
export async function saveImport(chunks: LibraryOp[][], deps: { apply: (ops: LibraryOp[]) => Promise<unknown>; sleep: (ms: number) => Promise<void>; onChunk: (n: number) => void }): Promise<'ok' | 'failed'>;
```

Order: all like ops, then for each playlist one `playlist_upsert` (name, origin) followed by its
`playlist_add` ops. Chunking: append ops to the current chunk while `ops.length < 100` and
`JSON.stringify({ ops, sentAt: 0 }).length + nextOpBytes <= 28_000`. `at` = `now` for all (the
merge rule is last-writer-wins by `at`, so a retry is a no-op). `saveImport`: 429 → sleep
Retry-After (default 10 s), retry the chunk, max 5; other error → 'failed'.
Tests: chunk limits on 1,000 likes with long titles; upsert before adds even across chunk
boundaries; deterministic playlist ids; already-liked excluded and counted; retry on 429.
