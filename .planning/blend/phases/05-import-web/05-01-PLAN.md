  <action>

```ts
import type { ImportBundle } from '@shared/importParse';

export interface MatchedTrack { readonly key: string; readonly song: SongSnapshot | null; readonly confidence: 'exact' | 'close' | 'none'; readonly accepted: boolean }
export type ImportState =
  | { readonly step: 'choose' }
  | { readonly step: 'reading'; readonly fileName: string }
  | { readonly step: 'preview'; readonly bundle: ImportBundle; readonly fileHash: string; readonly selectedPlaylists: ReadonlySet<string>; readonly includeLiked: boolean }
  | { readonly step: 'matching'; readonly bundle: ImportBundle; readonly fileHash: string; readonly done: number; readonly total: number; readonly results: ReadonlyMap<string, MatchedTrack> }
  | { readonly step: 'review'; readonly bundle: ImportBundle; readonly results: ReadonlyMap<string, MatchedTrack> }
  | { readonly step: 'saving'; readonly saved: number; readonly total: number }
  | { readonly step: 'done'; readonly added: number; readonly already: number; readonly playlists: number; readonly unmatched: number }
  | { readonly step: 'error'; readonly kind: 'file' | 'network'; readonly message: string; readonly resumable: boolean };

export type ImportAction =
  | { type: 'pick'; fileName: string } | { type: 'read'; bundle: ImportBundle; fileHash: string }
  | { type: 'togglePlaylist'; name: string } | { type: 'toggleLiked' } | { type: 'startMatching'; total: number }
  | { type: 'matched'; results: readonly [string, MatchedTrack][] } | { type: 'cancel' }
  | { type: 'finishMatching' } | { type: 'accept'; key: string; accepted: boolean } | { type: 'replace'; key: string; song: SongSnapshot }
  | { type: 'save'; total: number } | { type: 'savedChunk'; count: number }
  | { type: 'finish'; added: number; already: number; playlists: number; unmatched: number }
  | { type: 'fail'; kind: 'file' | 'network'; message: string; resumable: boolean } | { type: 'reset' };

export function importReducer(state: ImportState, action: ImportAction): ImportState;
```

Allowed transitions only (otherwise return `state` unchanged): choose→reading (pick);
reading→preview (read) | error(file); preview→matching (startMatching) | choose (reset);
matching→matching (matched) | review (finishMatching) | preview (cancel) | error(network, resumable);
review→review (accept, replace) | saving (save); saving→saving (savedChunk) | done (finish) |
error(network, resumable); error→choose (reset); done→choose (reset).
`key` = `identityKey(title, artist)` of the imported track. `close` matches start `accepted: true`,
`none` start `accepted: false`. Tests: every allowed transition, three disallowed ones.
