  bonus(userId: string, song: SongSnapshot, kind: 'like' | 'unlike' | 'playlistAdd', at: number): Promise<void>;

  top(userId: string, limit: number): Promise<readonly TallySong[]>;
  clear(userId: string): Promise<void>;
}
export class MemoryTasteTally implements TasteTally { /* same rules as convex/taste.ts, using addDecayed/listenAmount/TALLY_* from '../shared/blendDecay.js' and identityKey from '../shared/identity.js'; `now` injectable via constructor for tests */ }
