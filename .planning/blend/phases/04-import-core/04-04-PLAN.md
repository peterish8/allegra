```ts
export type MatchConfidence = 'exact' | 'close' | 'none';
export interface MatchResult { readonly index: number; readonly song: SongSnapshot | null; readonly confidence: MatchConfidence }
export const IMPORT_MATCH = { batchMax: 50, concurrency: 4, hitTtl: 30 * 86_400, missTtl: 7 * 86_400, durationToleranceSec: 5, closeSimilarity: 0.85 } as const;

/** Pure: how well a catalog row matches an imported track. */
export function scoreCandidate(track: ImportedTrack, song: UnifiedSong): MatchConfidence;
export class ImportMatcher {
  constructor(catalog: Pick<CatalogService, 'search'>, cache: CacheStore);
  match(tracks: readonly ImportedTrack[]): Promise<MatchResult[]>;
}
```

`scoreCandidate`: normalise titles with the same helper identityKey uses; lead artist =
`creditedArtists(x)[0]` lower-cased. exact: normalised titles equal AND lead artists equal AND
(no duration on either side OR |Δ| ≤ 5 s). close: titles equal and the artist sets intersect, or
Dice-coefficient bigram similarity of the normalised titles ≥ 0.85 with lead artists equal.
Otherwise none. Only Saavn rows (`song.source === 'Saavn'`) can match (D15).
`match`: for each track, `cachedLookup(cache, { key: cacheKey('import-match', identityKey(title, artist), String(duration ?? '')), hitTtlSeconds, missTtlSeconds, load })`;
`load` runs `catalog.search(`${title} ${leadArtist}`, 8, 1, { enrich: false })` inside try/catch
returning null on any error, picks the best candidate (exact over close; ties by catalog order),
returns `{ song: snapshot, confidence }` or null. Run loads through a 4-slot semaphore (write a
10-line `limit(4)` helper; no dependency). Cached value stores the snapshot and confidence.
Tests with a fake catalog: exact, close (feat. variant), none, Gaana-only result → none, cache hit
counts calls, throwing catalog → none, concurrency never above 4 (count in-flight in the fake).
