```ts
export const IMPORT_LIMITS = { tracks: 10_000, playlists: 200, textLength: 200 } as const;

export interface ImportedTrack {
  readonly title: string; readonly artist: string;
  readonly album?: string; readonly durationSec?: number;
}
export interface ImportedPlaylist { readonly name: string; readonly tracks: readonly ImportedTrack[] }
export interface ImportBundle {
  readonly source: 'spotify-export' | 'csv';
  readonly liked: readonly ImportedTrack[];
  readonly playlists: readonly ImportedPlaylist[];
  readonly skipped: number;      // podcasts, local files, malformed rows
  readonly truncated: boolean;   // a limit cut the input
}
export function parseSpotifyLibrary(json: unknown): { tracks: ImportedTrack[]; skipped: number };
export function parseSpotifyPlaylists(json: unknown): { playlists: ImportedPlaylist[]; skipped: number };
/** Library file plus every Playlist*.json, already JSON.parsed by the caller. */
export function bundleFromSpotifyFiles(files: { readonly name: string; readonly json: unknown }[]): ImportBundle;
export function parseCsv(text: string): ImportBundle;  // CSV rows become `liked`; no playlists
```

parser; no dependency); artists separated by ';' joined with ', '. Total tracks across liked and
playlists is capped at `IMPORT_LIMITS.tracks` in that order.
