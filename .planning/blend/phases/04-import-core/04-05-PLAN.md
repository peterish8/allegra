```ts
export const IMPORT_SEED_MAX_ARTISTS = 25;
/** Top artists of an import, log-scaled so the biggest counts SIGNAL_WEIGHT.seed. */
export function importSeedWeights(counts: readonly { name: string; count: number }[]): { name: string; weight: number }[] {
  const top = [...counts].filter((c) => c.count > 0 && c.name.trim()).sort((a, b) => b.count - a.count).slice(0, IMPORT_SEED_MAX_ARTISTS);
  const max = top[0]?.count ?? 0;
  return top.map((c) => ({ name: c.name.trim(), weight: max > 0 ? SIGNAL_WEIGHT.seed * Math.log2(1 + c.count) / Math.log2(1 + max) : 0 }));
}
export function applyImportSeed(taste: TasteProfile | undefined, counts: readonly { name: string; count: number }[], now = new Date()): TasteProfile;
```

`applyImportSeed` bumps each artist with its weight (reusing `bump`), leaves languages alone,
increments `signals` by the number applied, sets `onboarded: true`. Tests: 300 inputs → 25 applied;
top weight = 5; monotone weights; duplicate names merged case-insensitively before ranking.
